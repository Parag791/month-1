# Task 4: Netflix Titles Dataset EDA

import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# 1. Load dataset
df = pd.read_csv("netflix_titles.csv")

print("Dataset Shape:", df.shape)
print("\nFirst Five Rows:")
print(df.head())

print("\nMissing Values:")
print(df.isnull().sum())

# 2. Data cleaning
df["date_added"] = pd.to_datetime(
    df["date_added"].astype("string").str.strip(),
    errors="coerce"
)

for col in ["director", "cast", "country", "rating", "listed_in"]:
    df[col] = df[col].fillna("Unknown")

df = df.drop_duplicates(subset=["show_id"]).copy()

# 3. Feature engineering
df["release_year"] = pd.to_numeric(
    df["release_year"], errors="coerce"
)

df["release_decade"] = (
    df["release_year"] // 10 * 10
)

df["is_movie"] = (df["type"] == "Movie").astype(int)
df["date_added_year"] = df["date_added"].dt.year

sns.set_theme()

# 4. Movie vs TV Show pie chart
type_counts = df["type"].value_counts()

plt.figure(figsize=(6, 6))
plt.pie(
    type_counts.values,
    labels=type_counts.index,
    autopct="%1.1f%%",
    startangle=90
)
plt.title("Movies vs TV Shows")
plt.savefig("netflix_content_types.png", bbox_inches="tight")
plt.show()

# 5. Release year trend
year_counts = df.groupby("release_year")["show_id"].count()

plt.figure(figsize=(10, 5))
year_counts.plot()
plt.title("Netflix Titles by Release Year")
plt.xlabel("Release Year")
plt.ylabel("Number of Titles")
plt.savefig("netflix_release_trend.png", bbox_inches="tight")
plt.show()

# 6. Top 10 genres
genre_counts = (
    df["listed_in"].str.split(", ").explode()
    .loc[lambda x: x != "Unknown"]
    .value_counts().head(10)
)

plt.figure(figsize=(9, 5))
sns.barplot(x=genre_counts.values, y=genre_counts.index)
plt.title("Top 10 Netflix Genres")
plt.xlabel("Number of Titles")
plt.savefig("netflix_top_genres.png", bbox_inches="tight")
plt.show()

# 7. Top 5 countries
country_counts = (
    df["country"].str.split(", ").explode()
    .loc[lambda x: x != "Unknown"]
    .value_counts().head(5)
)

plt.figure(figsize=(8, 4))
sns.barplot(x=country_counts.values, y=country_counts.index)
plt.title("Top 5 Countries")
plt.xlabel("Number of Titles")
plt.savefig("netflix_top_countries.png", bbox_inches="tight")
plt.show()

# 8. Country vs release decade heatmap
top_countries = country_counts.index

country_data = df[["country", "release_decade"]].copy()
country_data["country"] = country_data["country"].str.split(", ")
country_data = country_data.explode("country")
country_data = country_data[
    country_data["country"].isin(top_countries)
]

pivot = pd.crosstab(
    country_data["country"],
    country_data["release_decade"]
)

plt.figure(figsize=(10, 5))
sns.heatmap(pivot, cmap="Blues")
plt.title("Country vs Release Decade")
plt.savefig("netflix_country_heatmap.png", bbox_inches="tight")
plt.show()

# 9. Save cleaned dataset
df.to_csv("netflix_titles_cleaned.csv", index=False)

# 10. Print insights
print("\nContent Type Counts:")
print(type_counts)

print("\nTop 10 Genres:")
print(genre_counts)

print("\nTop 5 Countries:")
print(country_counts)

print("\nTask 4 completed successfully!")
